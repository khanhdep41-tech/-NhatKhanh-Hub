local function queueForTeleport()

    local queuedCode = [[
        task.wait(2)

        local source =
            "https://raw.githubusercontent.com/khanhdep41-tech/-NhatKhanh-Hub/refs/heads/main/Auto%20Excute%20Hop%20Hub.lua"

        local success, result = pcall(function()

            local response = game:HttpGet(source)

            local fn = loadstring(response)

            if not fn then
                error("Không thể load source.")
            end

            fn()
        end)

        if not success then
            warn(
                "[Nhat Khan] Auto execute thất bại:",
                result
            )
        end
    ]]

    if type(queue_on_teleport) == "function" then

        local success = pcall(function()
            queue_on_teleport(queuedCode)
        end)

        if success then
            print(
                "[Nhat Khan] Đã queue auto execute."
            )
            return true
        end
    end

    if type(queueonteleport) == "function" then

        local success = pcall(function()
            queueonteleport(queuedCode)
        end)

        if success then
            print(
                "[Nhat Khan] Đã queue auto execute."
            )
            return true
        end
    end

    if type(syn) == "table"
        and type(syn.queue_on_teleport) == "function" then

        local success = pcall(function()
            syn.queue_on_teleport(queuedCode)
        end)

        if success then
            print(
                "[Nhat Khan] Đã queue auto execute."
            )
            return true
        end
    end

    warn(
        "[Nhat Khan] Executor không hỗ trợ queue_on_teleport."
    )

    return false
end

local function getServerPage(cursor)

    local url =
        "https://games.roblox.com/v1/games/"
        .. tostring(game.PlaceId)
        .. "/servers/Public?sortOrder=Asc&limit=100"

    if cursor and cursor ~= "" then

        url =
            url
            .. "&cursor="
            .. HttpService:UrlEncode(cursor)
    end

    local success, response =
        pcall(function()

            return game:HttpGet(url)

        end)

    if not success then

        return nil,
            "LỖI KẾT NỐI:\n"
            .. tostring(response)
    end

    local decodeSuccess, data =
        pcall(function()

            return HttpService:JSONDecode(
                response
            )
        end)

    if not decodeSuccess
        or type(data) ~= "table" then

        return nil,
            "Không thể đọc danh sách máy chủ."
    end

    if type(data.data) ~= "table" then

        return nil,
            "Không tìm thấy danh sách máy chủ."
    end

    return data
end

--==================================================
-- SHUFFLE
--==================================================

local function shuffle(list)

    for i = #list, 2, -1 do

        local j =
            math.random(1, i)

        list[i], list[j] =
            list[j], list[i]
    end

    return list
end

--==================================================
-- FIND 1-PLAYER SERVERS
--==================================================

local function findServers()

    local currentJobId =
        tostring(game.JobId)

    local candidates = {}
    local cursor = nil

    for page = 1, 10 do

        setStatus(
            "ĐANG TÌM MÁY CHỦ...\n"
            .. "Trang "
            .. page
            .. " / 10\n"
            .. "Đang tìm máy chủ có đúng 1 người"
        )

        local data, errorMessage =
            getServerPage(cursor)

        if not data then
            return nil, errorMessage
        end

        for _, server in
            ipairs(data.data) do

            local id =
                tostring(server.id or "")

            local playing =
                tonumber(server.playing or 0)

            local maxPlayers =
                tonumber(server.maxPlayers or 0)

            if id ~= ""
                and id ~= currentJobId
                and playing == 1
                and maxPlayers > 1 then

                local duplicate = false

                for _, existing in
                    ipairs(candidates) do

                    if existing.id == id then
                        duplicate = true
                        break
                    end
                end

                if not duplicate then

                    table.insert(
                        candidates,
                        {
                            id = id,
                            playing = playing,
                            maxPlayers = maxPlayers
                        }
                    )
                end
            end
        end

        if #candidates >= 20 then
            break
        end

        cursor =
            data.nextPageCursor

        if not cursor
            or cursor == "" then

            break
        end

        task.wait(0.15)
    end

    if #candidates == 0 then

        return nil,
            "Không tìm thấy máy chủ công khai có đúng 1 người."
    end

    shuffle(candidates)

    return candidates
end

local function tryTeleport(server)

    teleportFailed = false
    teleportFailureMessage = ""

    setStatus(
        "ĐANG CHUYỂN MÁY CHỦ...\n"
        .. "Người chơi: "
        .. tostring(server.playing)
        .. " / "
        .. tostring(server.maxPlayers)
    )

    queueForTeleport()

    local success, errorMessage =
        pcall(function()

            TeleportService:
                TeleportToPlaceInstance(
                    game.PlaceId,
                    server.id,
                    LocalPlayer
                )
        end)

    if not success then

        return false,
            tostring(errorMessage)
    end

    for i = 1, 50 do

        if teleportFailed then

            return false,
                teleportFailureMessage
        end

        task.wait(0.1)
    end

    if game.JobId == server.id then
        return true
    end

    return false,
        "Không thể hoàn tất chuyển máy chủ."
end

-- HOP
--==================================================

local function hop()

    if hopping then
        return
    end

    hopping = true

    HopButton.Text =
        "ĐANG TÌM..."

    HopButton.Active = false

    local servers, errorMessage =
        findServers()

    if not servers then

        setStatus(
            "KHÔNG TÌM THẤY MÁY CHỦ\n\n"
            .. tostring(errorMessage)
        )

        HopButton.Text =
            "CHUYỂN NGAY"

        HopButton.Active = true
        hopping = false

        return
    end

    local attempts =
        math.min(#servers, 8)

    for attempt = 1, attempts do

        local server =
            servers[attempt]

        setStatus(
            "ĐANG THỬ MÁY CHỦ "
            .. attempt
            .. " / "
            .. attempts
            .. "\n"
            .. "Người chơi: "
            .. server.playing
            .. " / "
            .. server.maxPlayers
        )

        local success, reason =
            tryTeleport(server)

        if success then

            setStatus(
                "ĐANG CHUYỂN...\n\n"
                .. "Đang vào máy chủ 1 người."
            )

            return
        end

        warn(
            "[Nhat Khan] Lần thử "
            .. attempt
            .. " thất bại: "
            .. tostring(reason)
        )

        task.wait(0.5)
    end

    setStatus(
        "TẤT CẢ LẦN THỬ ĐỀU THẤT BẠI\n\n"
        .. "Hãy thử lại."
    )

    HopButton.Text =
        "CHUYỂN NGAY"

    HopButton.Active = true
    hopping = false
end

--==================================================

local function startAutoHop()

    if not autoHop then
        return
    end

    if autoLoopRunning then
        return
    end

    autoLoopRunning = true

    task.spawn(function()

        task.wait(1)

        while autoHop do

            -- Wait for any existing hop
            while hopping and autoHop do
                task.wait(0.25)
            end

            if not autoHop then
                break
            end

            -- Start a hop
            task.spawn(hop)

            -- IMPORTANT:
            -- Wait until hop() actually finishes
            while hopping and autoHop do
                task.wait(0.25)
            end

            if not autoHop then
                break
            end

            -- Delay before next hop
            for i = 1, 5 do

                if not autoHop then
                    break
                end

                task.wait(1)
            end
        end

        autoLoopRunning = false
    end)
end
